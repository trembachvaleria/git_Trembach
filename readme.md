git init - инициализация репрозитория

git commit - фиксация изменений

git status - показывает текущее состояние репрозитория

git add - добавить файл

git branch - управление ветками в репрозитории

git switch - переключение между ветками

git push - отправляет локальные изменения из репрозитория в удаленные

git merge - слить указанную ветку в текущую

git branch -d - удаляет локальную ветку

git checkout - переключение между ветками

git log - посмотреть историю коммитов
